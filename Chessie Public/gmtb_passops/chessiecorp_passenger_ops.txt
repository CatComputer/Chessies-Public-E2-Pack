#[ 

instructions:
    - grab configs from github under gmtb_passops (works the same as PT2)
    - set the include above to use your current map
    - wire EGP:wirelink to any EGP you want to use as a status board
    - refresh the E2
    - ah, you're done already

how to use:
    - all inputs below require the person to use their stunstick. these keybinds work for every player
    
    - E: initial setup for car
    - left-click: increase car capacity
    - right-click: decrease car capacity
    - E after setup: prepares car for running
    - R: remove car (press again to immediately terminate operation)
    
    
    easy! (maybe)







table data:
    UserData - 
        steamid64 - 
            1: name [string]
            2: cars [table]
            3: money [number]
            4: passengers onboard [number]
            5: passengers serviced [number]
            6: number of cars [number]

        

    CarTable - 
        car_ent_id - 
            1: car entity [entity]
            2: capacity [number]
            3: car ready to enroll [number]
            4: steamid64 owner [string]
            5: car enrolled [number]
            6: car in station [number]
            7: mileage [number]
            8: seconds elapsed [number]
            9: passengers [number]
            10: last station [string]
            11: last speed [number]
            12: happiness [number]
            13: terminate on next dropoff [number]
            14: default capacity [number]
            
    Stations - 
        station_id -
            name [string]
            name_position [vector]
            trigger_start [vector]
            trigger_end [vector]
            capacity [number]
            occupy_rate [number]
            people [number]
            people_postload [number]

]#

function void wirelink:playerData(I:number,POS:vector2,DATA:table){
    local E=This
    
    E:egpTextLayout(I,format("%s  |  $%.02f  |  %01d car(s)\n%01d/%01d pass. onboard/serviced",DATA[1,string],DATA[3,number],DATA[6,number],DATA[4,number],DATA[5,number]),POS,vec2(300,40))
    E:egpAlign(I,0,1)
    E:egpFont(I,"Arial",11)
    E:egpColor(I,vec(20,22,24))
}

function void wirelink:stationData(I:number,POS:vector2,DATA:table){
    local E=This
    
    E:egpCircle(I,POS,vec2(8))
    E:egpColor(I,vec(255-(DATA["people",number]/DATA["capacity",number])*255,(DATA["people",number]/DATA["capacity",number])*255,0))
    
    E:egpCircleOutline(I+1,POS,vec2(12))
    E:egpColor(I+1,vec(20,22,24))
    
    E:egpTextLayout(I+2,format("%s\n%01d / %01d pass. waiting",DATA["name",string],DATA["people",number],DATA["capacity",number]),POS+vec2(130,0),vec2(220,40))
    E:egpAlign(I+2,0,1)
    E:egpFont(I+2,"Arial",11)
    E:egpColor(I+2,vec(20,22,24))
}

if(Run){
    
    #[timer("station_setup",0.1,0,function(){
        if(Station<Stations:count()){

    timer("clk",1,0,function(){
        #passops portion
        
        #station lookup and person update
        for(I=1,Stations:count()){
            local Station = Stations[I,table]
            
            Stations[I,table]["people",number] = Station["people",number] + Station["occupy_rate",number]
            
            Stations[I,table]["people_postload",number] = Station["people",number]
            
            Stations[I,table]["people",number] = clamp(Station["people",number],0,Station["capacity",number])
        }
        
        #car scanning loop
        for(I=1,CarTable:count()){
            local ID = CarTable:keys()[I,number]
            
            local Car = CarTable[ID,table]
            
            
            #initialized check
            if(Car[3,number]){
                CarTable[ID,table][5,number] = 1
                
                local E = Car[1,entity]
                
                if(E:isValid()){
                    #variable declaration
                    local Cap = Car[2,number]
                    local Speed = toUnit("mph",max(abs(Car[1,entity]:velL():x()),abs(Car[1,entity]:velL():y()),abs(Car[1,entity]:velL():z())))
                    local Pos = E:pos()
                    
                    local Station = ""
                    local SID = 0
                    
                    #speed and mileage update
                    if(CarTable[ID,table][9,number]){
                        CarTable[ID,table][7,number] = (Car[7,number]+Speed/3600)
                        
                        CarTable[ID,table][8,number] = Car[8,number] + 1
                    }
                    
                    #local AvgSpeed = (Car[7,number]/max(Car[8,number],1))*3600
                    
                    #accelerometer
                    local Acceleration = Speed - Car[11,number]
                    
                    CarTable[ID,table][11,number] = Speed
                    
                    
                    #happiness system
                    local PassiveGain = CFG["happiness_passive_gain",number]*(1-Car[12,number])
                    
                    local AccelLoss = CFG["happiness_acceleration_loss",number]*abs(Acceleration)
                    
                    local LowSpeedLoss = CFG["happiness_low_speed_loss",number] * max(CFG["happiness_low_speed_threshold",number]-Speed,0)
                    
                    local HighSpeedLoss = CFG["happiness_high_speed_loss",number] * max(Speed-CFG["happiness_high_speed_threshold",number],0)
                    
                
                    CarTable[ID,table][12,number] = Car[12,number] + (PassiveGain - AccelLoss - LowSpeedLoss - HighSpeedLoss)*CFG["happiness_speed_multiplier",number]
                    
                    CarTable[ID,table][12,number] = clamp(Car[12,number],0,1)
                    
                    CarTable[ID,table][6,number] = 0
                    
                    
                    #in-station checking
                    for(J=1,Stations:count()){
                        local S = Stations[J,table]
                        
                        if(inrange(Pos,S["trigger_start",vector],S["trigger_end",vector])){
                            CarTable[ID,table][6,number] = 1
                            Station = S["name",string]
                            SID = J
                        }
                    }
                    
                    local InStation = Car[6,number]
                    
                    #load/unload system
                    
                    if(InStation&(Station!=(Car[10,string]))&SID){
                        if(Speed>1){
                            E:setCarTag(format("CONDUCTOR %s\nPASSENGER COUNT - %01d/%01d\nSTATION - %s\nTOO FAST TO EXCHANGE",findPlayerBySteamID64(Car[4,string]):name():upper(),Car[9,number],Cap,Station:upper()),vec(220,160,70),vec(20,30,40))
                            
                            if(timerExists(ID:toString())){
                                stoptimer(ID:toString())
                            }
                        }
                        else{
                            local RNG = random(0.5,1)
                            
                            local ProfitEst = Car[9,number]*CFG["fare_per_mile",number]*Car[7,number]*(Car[12,number])

                                E:setCarTag(format("CONDUCTOR %s\nPASSENGER COUNT - %01d/%01d\nSTATION - %s\nAWAITING PASSENGERS\nESTIMATED PROFIT - $%.02f\nPENALTY DEDUCTION - %01d%%",findPlayerBySteamID64(Car[4,string]):name():upper(),Car[9,number],Cap,Station:upper(),
                                ProfitEst,100-Car[12,number]*100),vec(220,160,70),vec(20,30,40))
                            }
                            else{
                                E:setCarTag(format("CONDUCTOR %s\nPASSENGER COUNT - %01d/%01d\nSTATION - %s\nAWAITING PASSENGERS",findPlayerBySteamID64(Car[4,string]):name():upper(),Car[9,number],Cap,Station:upper()),
                                vec(220,160,70),vec(20,30,40))
                            }
                            
                            Stations[SID,table]["people_postload",number] = Stations[SID,table]["people_postload",number] - Cap
                            
                            if(!timerExists(ID:toString()) & Car[3,number]){
                                local IDF = ID
                                local RNGF = RNG
                                local ProfitEstF = ProfitEst
                                
                                timer(IDF:toString(),CFG["wait_time",number],1,function(){
                                    
                                    
                                    
                                    UserData[CarTable[IDF,table][4,string],table][3,number] = UserData[CarTable[IDF,table][4,string],table][3,number] + ProfitEstF
                                    
                                    UserData[CarTable[IDF,table][4,string],table][4,number] = UserData[CarTable[IDF,table][4,string],table][4,number] - CarTable[IDF,table][9,number]
                                    UserData[CarTable[IDF,table][4,string],table][5,number] = UserData[CarTable[IDF,table][4,string],table][5,number] + CarTable[IDF,table][9,number]
                                    
                                    CarTable[IDF,table][7,number] = 0
                                    CarTable[IDF,table][8,number] = 0
                                    
                                    if(!CarTable[IDF,table][13,number]){
                                    
                                        CarTable[IDF,table][9,number] = floor(clamp(floor(Stations[SID,table]["people",number]),0,Cap)*RNGF)
                                        
                                        UserData[CarTable[IDF,table][4,string],table][4,number] = UserData[CarTable[IDF,table][4,string],table][4,number] + CarTable[IDF,table][9,number]
                                        
                                        Stations[SID,table]["people",number] = Stations[SID,table]["people",number] - floor(clamp(floor(Stations[SID,table]["people",number]),0,Cap)*RNGF)
                                        
                                        CarTable[IDF,table][6,number] = 0
                                        
                                        CarTable[IDF,table][12,number] = 0.75
                                        
                                        
                                        
                                        CarTable[IDF,table][10,string] = Station
                                    }
                                    else{
                                        UserData[CarTable[IDF,table][4,string],table][6,number] = UserData[E:steamID64(),table][6,number] - 1
                                        
                                        UserData[CarTable[IDF,table][4,string],table][2,table][IDF,table] = table()
                                        
                                        UserData[CarTable[IDF,table][4,string],table][2,table]:remove(IDF)
                                        
                                        CarTable[IDF,table][1,entity]:clearCarTag()
                        
                                        CarTable[IDF,table] = table()
                                        
                                        CarTable:remove(IDF)
                                    }
                                })
                            }
                        }
                    }
                    else{
                        if(Car[9,number]){
                            if(Car[13,number]){
                                E:setCarTag(format("CONDUCTOR %s\nPASSENGER COUNT - %01d/%01d\nMILEAGE - %.02f\nTIME IN TRANSIT - %01d:%02d\nPASSENGER HAPPINESS - %.01f%%\nCAR SET TO TERMINATE AT NEXT STOP\nPRESS 'R' AGAIN TO FORCE TERMINATION",

                                floor(CarTable[CarTable:keys()[I,number],table][8,number]/60),CarTable[CarTable:keys()[I,number],table][8,number]%60,Car[12,number]*100),vec(220,60,40),vec(20,30,40))
                            }
                            else{
                                E:setCarTag(format("CONDUCTOR %s\nPASSENGER COUNT - %01d/%01d\nMILEAGE - %.02f\nTIME IN TRANSIT - %01d:%02d\nPASSENGER HAPPINESS - %.01f%%",findPlayerBySteamID64(Car[4,string]):name():upper(),Car[9,number],Cap,CarTable[CarTable:keys()[I,number],table][7,number],
                                floor(CarTable[CarTable:keys()[I,number],table][8,number]/60),CarTable[CarTable:keys()[I,number],table][8,number]%60,Car[12,number]*100),vec(90,140,240),vec(20,30,40))

                        }

event keyPressed(E:entity,K:string,D:number,KB:string){
    if(E & (E:weapon()) & (E:weapon():getClass()=="weapon_stunstick")){
        
        if(K=="mouse_left"&D){
            local C = E:aimEntity()
            
            if(C&(C:getClass()=="prop_physics")&(C:mass()>1000)){
                local VibeCheck = ((findPlayerBySteamID64(CarTable[C:id(),table][4,string]))==E)|(E==owner())
                
                if(CarTable[C:id(),table][1,entity]&VibeCheck&(!CarTable[C:id(),table][3,number])){
                    if(CarTable[C:id(),table][2,number]>=20){
                        CarTable[C:id(),table][2,number] = CarTable[C:id(),table][2,number] + 5
                        
                        CarTable[C:id(),table][2,number] = round(CarTable[C:id(),table][2,number]/5,0)*5
                    }
                    else{
                        CarTable[C:id(),table][2,number] = CarTable[C:id(),table][2,number] + 1
                    }
                    
                    CarTable[C:id(),table][2,number] = clamp(CarTable[C:id(),table][2,number],1,CarTable[C:id(),table][14,number]*2)
                    
                    C:setCarTag(format("CAR SET UP\nCAPACITY: %01d PEOPLE",CarTable[C:id(),table][2,number]),vec(220,60,40),vec(20,30,40))
                }
            }
        }
        
        if(K=="mouse_right"&D){
            local C = E:aimEntity()
            
            if(C&(C:getClass()=="prop_physics")&(C:mass()>1000)){
                local VibeCheck = ((findPlayerBySteamID64(CarTable[C:id(),table][4,string]))==E)|(E==owner())
                
                if(CarTable[C:id(),table][1,entity]&VibeCheck&(!CarTable[C:id(),table][3,number])){
                    if(CarTable[C:id(),table][2,number]>20){
                        CarTable[C:id(),table][2,number] = CarTable[C:id(),table][2,number] - 5
                        
                        CarTable[C:id(),table][2,number] = round(CarTable[C:id(),table][2,number]/5,0)*5
                    }
                    else{
                        CarTable[C:id(),table][2,number] = CarTable[C:id(),table][2,number] - 1
                    }
                    
                    CarTable[C:id(),table][2,number] = clamp(CarTable[C:id(),table][2,number],1,CarTable[C:id(),table][14,number]*2)
                    
                    C:setCarTag(format("CAR SET UP\nCAPACITY: %01d PEOPLE",CarTable[C:id(),table][2,number]),vec(220,60,40),vec(20,30,40))
                }
            }
        }
        
        if(K=="e"&D){
            local C = E:aimEntity()
            
            if(C&(C:getClass()=="prop_physics")&(C:mass()>1000)){
                local VibeCheck = ((findPlayerBySteamID64(CarTable[C:id(),table][4,string]))==E)|(E==owner())
                
                if(!CarTable[C:id(),table][1,entity]){
                    local Box = C:boxSize()
                    local CarSize = max(Box:x(),Box:y(),Box:z())/7
                    
                    CarTable[C:id(),table] = table(C,CarSize,0,E:steamID64(),0,0,0,0,0,"",0,0,0,CarSize)
                    
                    C:setCarTag(format("CAR SET UP\nCAPACITY: %01d PEOPLE",CarTable[C:id(),table][2,number]),vec(220,60,40),vec(20,30,40))
                }
                
                elseif(VibeCheck&(!CarTable[C:id(),table][3,number])){
                    CarTable[C:id(),table][3,number] = 1
                    
                    if(!UserData[E:steamID64(),table]){
                        UserData[E:steamID64(),table] = table(E:name(),table(),0,0,0,0)
                    }
                    UserData[E:steamID64(),table][2,table][C:id(),table] = CarTable[C:id(),table]
                    
                    UserData[E:steamID64(),table][6,number] = UserData[E:steamID64(),table][6,number] + 1
                    
                    C:setCarTag(format("SETUP COMPLETE\nCAPACITY OF %01d PEOPLE\nAWAITING CLASSIFICATION",CarTable[C:id(),table][2,number]),vec(40,220,60),vec(20,30,40))
                }
            }
        }
        
        if(K=="r"&D){
            local C = E:aimEntity()
            
            if(C&(C:getClass()=="prop_physics")&(C:mass()>1000)){
                local VibeCheck = ((findPlayerBySteamID64(CarTable[C:id(),table][4,string]))==E)|(E==owner())
                
                if((CarTable[C:id(),table][1,entity]|C:readCarTag())&VibeCheck){
                    
                    if((!CarTable[C:id(),table][13,number])&(CarTable[C:id(),table][9,number])){
                        CarTable[C:id(),table][13,number] = 1
                    }
                    else{
                        UserData[E:steamID64(),table][4,number] = UserData[E:steamID64(),table][4,number] - CarTable[C:id(),table][9,number]
                    
                        UserData[E:steamID64(),table][6,number] = UserData[E:steamID64(),table][6,number] - 1
                        
                        CarTable[C:id(),table] = table()
                        
                        CarTable:remove(C:id())
                        
                        UserData[E:steamID64(),table][2,table][C:id(),table] = table()
                        
                        UserData[E:steamID64(),table][2,table]:remove(C:id())
                    }
                    
                    
                    
                    C:clearCarTag()
                }
            }
        }
    }
}
